			((m : mem x a) => false.elim (mem x b) (ha x m))

pair : Set -> Set -> Set :=
	(a : Set) => (b : Set) =>
	collect ((x : Set) => or (eq Set x a) (eq Set x b)) (pairing a b);

mem_pair : (a : Set) -> (b : Set) -> (x : Set) ->

union : Set -> Set :=
	(a : Set) =>
	collect ((x : Set) => exists Set ((y : Set) => and (mem y a) (mem x y)))
		(union_exists a);

mem_union : (a : Set) -> (x : Set) ->

	(a : Set) =>

power : Set -> Set :=

separate : (Set -> Prop) -> Set -> Set :=
	(P : Set -> Prop) => (a : Set) =>
	collect ((x : Set) => and (mem x a) (P x)) (separation P a);
