Nat.lt_double : (m : Nat) -> Nat.Lt (Nat.succ m) (Nat.double (Nat.succ m)) :=
	(m : Nat) =>
	Exists.intro Nat
		((j : Nat) =>
			Eq Nat (Nat.add (Nat.succ (Nat.succ m)) j) (Nat.double (Nat.succ m)))
		m
		(Eq.trans Nat
			(Nat.add (Nat.succ (Nat.succ m)) m)
			(Nat.succ (Nat.add (Nat.succ m) m))
			(Nat.succ (Nat.succ (Nat.double m)))
			(Nat.add.succ_add (Nat.succ m) m)
			(Eq.congr Nat Nat Nat.succ
				(Nat.add (Nat.succ m) m)
				(Nat.succ (Nat.double m))
				(Eq.trans Nat
					(Nat.add (Nat.succ m) m)
					(Nat.succ (Nat.add m m))
					(Nat.succ (Nat.double m))
					(Nat.add.succ_add m m)
					(Eq.congr Nat Nat Nat.succ (Nat.add m m) (Nat.double m)
						(Eq.symm Nat (Nat.double m) (Nat.add m m) (Nat.double_eq m))))));

Nat.NotSqrt2Denom : Nat -> Prop :=
	(b : Nat) => (a : Nat) ->
		Eq Nat (Nat.mul a a) (Nat.double (Nat.mul b b)) -> Eq Nat b Nat.zero;

Nat.not_sqrt2_denom_step : (b : Nat) -> ((m : Nat) -> Nat.Lt m b -> Nat.NotSqrt2Denom m) -> Nat.NotSqrt2Denom b :=
	(b : Nat) =>
	(ih : (m : Nat) -> Nat.Lt m b -> Nat.NotSqrt2Denom m) =>
	(a : Nat) =>
	(h : Eq Nat (Nat.mul a a) (Nat.double (Nat.mul b b))) =>
	Exists.elim Nat ((k : Nat) => Eq Nat a (Nat.double k))
		(Eq Nat b Nat.zero)
		(Nat.double_of_sq_eq_double a (Nat.mul b b) h)
		((k : Nat) => (ha : Eq Nat a (Nat.double k)) =>
			Exists.elim Nat ((m : Nat) => Eq Nat b (Nat.double m))
				(Eq Nat b Nat.zero)
				(Nat.double_of_sq_eq_double b (Nat.mul k k)
					(Eq.symm Nat (Nat.double (Nat.mul k k)) (Nat.mul b b)
						(Nat.double_inj (Nat.double (Nat.mul k k)) (Nat.mul b b)
							(Eq.trans Nat
								(Nat.double (Nat.double (Nat.mul k k)))
								(Nat.mul (Nat.double k) (Nat.double k))
								(Nat.double (Nat.mul b b))
								(Eq.symm Nat
									(Nat.mul (Nat.double k) (Nat.double k))
									(Nat.double (Nat.double (Nat.mul k k)))
									(Nat.double_sq k))
								(Eq.subst Nat a (Nat.double k)
									((t : Nat) => Eq Nat (Nat.mul t t) (Nat.double (Nat.mul b b)))
									ha h)))))
				((m : Nat) => (hb : Eq Nat b (Nat.double m)) =>
					Nat.cases
						((t : Nat) =>
							Eq Nat b (Nat.double t) ->
							Eq Nat (Nat.mul k k) (Nat.double (Nat.mul t t)) ->
							Eq Nat b Nat.zero)
						((hb0 : Eq Nat b Nat.zero) =>
							(e : Eq Nat (Nat.mul k k) Nat.zero) => hb0)
						((m1 : Nat) =>
							(hb1 : Eq Nat b (Nat.double (Nat.succ m1))) =>
							(e : Eq Nat (Nat.mul k k)
								(Nat.double (Nat.mul (Nat.succ m1) (Nat.succ m1)))) =>
							False.elim (Eq Nat b Nat.zero)
								(Nat.succ_ne_zero m1
									(ih (Nat.succ m1)
										(Eq.subst Nat (Nat.double (Nat.succ m1)) b
											((t : Nat) => Nat.Lt (Nat.succ m1) t)
											(Eq.symm Nat b (Nat.double (Nat.succ m1)) hb1)
											(Nat.lt_double m1))
										k e)))
						m
						hb
						(Eq.symm Nat
							(Nat.double (Nat.mul m m))
							(Nat.mul k k)
							(Nat.double_inj (Nat.double (Nat.mul m m)) (Nat.mul k k)
								(Eq.trans Nat
									(Nat.double (Nat.double (Nat.mul m m)))
									(Nat.mul b b)
									(Nat.double (Nat.mul k k))
									(Eq.symm Nat
										(Nat.mul b b)
										(Nat.double (Nat.double (Nat.mul m m)))
										(Eq.trans Nat
											(Nat.mul b b)
											(Nat.mul (Nat.double m) (Nat.double m))
											(Nat.double (Nat.double (Nat.mul m m)))
											(Eq.subst Nat b (Nat.double m)
												((t : Nat) => Eq Nat (Nat.mul b b) (Nat.mul t t))
												hb (Eq.refl Nat (Nat.mul b b)))
											(Nat.double_sq m)))
									(Eq.symm Nat
										(Nat.double (Nat.mul k k))
										(Nat.mul b b)
										(Nat.double_inj (Nat.double (Nat.mul k k)) (Nat.mul b b)
											(Eq.trans Nat
												(Nat.double (Nat.double (Nat.mul k k)))
												(Nat.mul (Nat.double k) (Nat.double k))
												(Nat.double (Nat.mul b b))
												(Eq.symm Nat
													(Nat.mul (Nat.double k) (Nat.double k))
													(Nat.double (Nat.double (Nat.mul k k)))
													(Nat.double_sq k))
												(Eq.subst Nat a (Nat.double k)
													((t : Nat) => Eq Nat (Nat.mul t t) (Nat.double (Nat.mul b b)))
													ha h)))))))));

// a^2 = 2 b^2 -> b = 0
Nat.sq_eq_double_sq_zero : (a : Nat) -> (b : Nat) ->
	Eq Nat (Nat.mul a a) (Nat.double (Nat.mul b b)) -> Eq Nat b Nat.zero :=
	(a : Nat) => (b : Nat) =>
	Nat.strong_ind Nat.NotSqrt2Denom Nat.not_sqrt2_denom_step b a;

// a^2 = 2b^2 -> b^2 = 0
// a = sqrt(2) * b^2 -> b^2 = 0 
// wait a minute this doesn't prove sqrt(2) is irrational
// rationals aren't even defined!!
// god fucking dammit
Nat.sqrt2_irrational : (a : Nat) -> (b : Nat) ->
	Eq Nat (Nat.mul a a) (Nat.mul Nat.two (Nat.mul b b)) -> Eq Nat b Nat.zero :=
	(a : Nat) => (b : Nat) =>
	(h : Eq Nat (Nat.mul a a) (Nat.mul Nat.two (Nat.mul b b))) =>
	Nat.sq_eq_double_sq_zero a b
		(Eq.trans Nat
			(Nat.mul a a)
			(Nat.mul Nat.two (Nat.mul b b))
			(Nat.double (Nat.mul b b))
			h
			(Nat.mul.two_mul (Nat.mul b b)));

// b != 0 -> a^2 != 2 * b^2
Nat.sqrt2_irrational_ne : (a : Nat) -> (b : Nat) ->
	NotEq Nat b Nat.zero ->
	NotEq Nat (Nat.mul a a) (Nat.mul Nat.two (Nat.mul b b)) :=
	(a : Nat) => (b : Nat) =>
	(hb : NotEq Nat b Nat.zero) =>
	(h : Eq Nat (Nat.mul a a) (Nat.mul Nat.two (Nat.mul b b))) =>
	hb (Nat.sqrt2_irrational a b h);
