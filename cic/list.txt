@require "nat.txt"

@inductive
List : Type -> Type;
@intro(List)
List.nil : (El : Type) -> List El;
@intro(List)
List.cons : (El : Type) -> (head : El) -> (tail : List El) -> List El;

@rec(List)
List.rec :
	(El : Type) ->
	(motive : List El -> Type) ->
	(case_nil : motive (List.nil El)) ->
	(case_cons : (head : El) -> (tail : List El) -> motive tail ->
		motive (List.cons El head tail)) ->
	(target : List El) ->
	motive target;

@iota(List)
List.rec El motive case_nil case_cons (List.nil El) :
	motive (List.nil El) := case_nil;
@iota(List)
List.rec El motive case_nil case_cons (List.cons El head tail) :
	motive (List.cons El head tail) :=
	case_cons head tail (List.rec El motive case_nil case_cons tail);

List.len : (El : Type) -> (xs : List El) -> Nat :=
	(El : Type) => (xs : List El) =>
	List.rec El
		((t : List El) => Nat)
		Nat.zero
		((head : El) => (tail : List El) => (ih : Nat) => Nat.succ ih)
		xs;
// Proof that len([]) = 0
List.len.nil : (El : Type) -> Eq Nat (List.len El (List.nil El)) Nat.zero :=
	(El : Type) => Eq.refl Nat Nat.zero;
// Proof that len([head, ...tail]) = succ(len(tail))
List.len.cons : (El : Type) -> (head: El) -> (tail : List El) ->
	Eq Nat (List.len El (List.cons El head tail)) (Nat.succ (List.len El tail)) :=
	(El : Type) => (head : El) => (tail : List El) =>
	Eq.refl Nat (Nat.succ (List.len El tail));

List.join : (El : Type) -> (a : List El) -> (b : List El) -> List El :=
	(El : Type) => (a : List El) => (b : List El) =>
	List.rec El
		((t : List El) => List El)
		b
		((head : El) => (tail : List El) => (ih : List El) =>
			List.cons El head ih)
		a;
// Join([], a) = a
List.join.nil : (El : Type) -> (a : List El) ->
	Eq (List El) (List.join El (List.nil El) a) a :=
	(El : Type) => (a : List El) => Eq.refl (List El) a;
// Join([head, ...tail], a) = [head, ...Join(tail, a)]
List.join.cons : (El : Type) -> (head : El) -> (tail: List El) -> (a : List El) ->
	Eq (List El) (List.join El (List.cons El head tail) a) (List.cons El head (List.join El tail a)) :=
	(El : Type) => (head : El) => (tail : List El) => (a : List El) =>
	Eq.refl (List El) (List.cons El head (List.join El tail a));
