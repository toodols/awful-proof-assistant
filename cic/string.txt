Char.valid : Nat -> Prop :=
        (n : Nat) => Prop.And
                (Nat.Lt n 1114112)
                (Prop.Not (Prop.And (Nat.Le 55296 n) (Nat.Le n 57343)));

@inductive
Char : Type;

@intro(Char)
@lang(Char.mk)
Char.intro : (n : Nat) -> Char.valid n -> Char;
@rec(Char)
Char.rec :
        (motive : Char -> Type) ->
        (case_intro : (n : Nat) -> (h : Char.valid n) -> motive (Char.intro n h)) ->
        (target : Char) ->
        motive target;
	
@iota(Char)
Char.rec motive case_intro (Char.intro n h) : motive (Char.intro n h) := case_intro n h;

// Char.valid n, decided by computing: below the maximum, and not in the
// surrogate range. Each comparison is a Nat.dec_le, which runs Nat.ble.
Char.dec_valid : (n : Nat) -> Decidable (Char.valid n) :=
	(n : Nat) =>
	Decidable.rec (Nat.Lt n 1114112)
		((t : Decidable (Nat.Lt n 1114112)) => Decidable (Char.valid n))
		((hlt : Nat.Lt n 1114112) =>
			Decidable.rec (Nat.Le 55296 n)
				((t : Decidable (Nat.Le 55296 n)) => Decidable (Char.valid n))
				((h1 : Nat.Le 55296 n) =>
					Decidable.rec (Nat.Le n 57343)
						((t : Decidable (Nat.Le n 57343)) => Decidable (Char.valid n))
						((h2 : Nat.Le n 57343) =>
							Decidable.no (Char.valid n)
								((hv : Char.valid n) =>
									Prop.And.right (Nat.Lt n 1114112)
										(Prop.Not (Prop.And (Nat.Le 55296 n) (Nat.Le n 57343))) hv
										(Prop.And.intro (Nat.Le 55296 n) (Nat.Le n 57343) h1 h2)))
						((nh2 : Prop.Not (Nat.Le n 57343)) =>
							Decidable.yes (Char.valid n)
								(Prop.And.intro (Nat.Lt n 1114112)
									(Prop.Not (Prop.And (Nat.Le 55296 n) (Nat.Le n 57343)))
									hlt
									((hand : Prop.And (Nat.Le 55296 n) (Nat.Le n 57343)) =>
										nh2 (Prop.And.right (Nat.Le 55296 n) (Nat.Le n 57343) hand))))
						(Nat.dec_le n 57343))
				((nh1 : Prop.Not (Nat.Le 55296 n)) =>
					Decidable.yes (Char.valid n)
						(Prop.And.intro (Nat.Lt n 1114112)
							(Prop.Not (Prop.And (Nat.Le 55296 n) (Nat.Le n 57343)))
							hlt
							((hand : Prop.And (Nat.Le 55296 n) (Nat.Le n 57343)) =>
								nh1 (Prop.And.left (Nat.Le 55296 n) (Nat.Le n 57343) hand))))
				(Nat.dec_le 55296 n))
		((nhlt : Prop.Not (Nat.Lt n 1114112)) =>
			Decidable.no (Char.valid n)
				((hv : Char.valid n) =>
					nhlt (Prop.And.left (Nat.Lt n 1114112)
						(Prop.Not (Prop.And (Nat.Le 55296 n) (Nat.Le n 57343))) hv)))
		(Nat.dec_le (Nat.succ n) 1114112);

// A character from a number, if the number is a valid code point. Decided by Char.dec_valid,
// so it computes; the two lemmas below say what it computes.
Char.try_of_nat : Nat -> Option Char :=
	(n : Nat) =>
	Decidable.rec (Char.valid n)
		((t : Decidable (Char.valid n)) => Option Char)
		((h : Char.valid n) => Option.some Char (Char.intro n h))
		((h : Prop.Not (Char.valid n)) => Option.none Char)
		(Char.dec_valid n);

Char.try_of_nat_some : (n : Nat) -> (h : Char.valid n) ->
	Eq (Option Char) (Char.try_of_nat n) (Option.some Char (Char.intro n h)) :=
	(n : Nat) => (h : Char.valid n) =>
	Decidable.rec_prop (Char.valid n)
		((t : Decidable (Char.valid n)) =>
			Eq (Option Char)
				(Decidable.rec (Char.valid n)
					((s : Decidable (Char.valid n)) => Option Char)
					((h2 : Char.valid n) => Option.some Char (Char.intro n h2))
					((h2 : Prop.Not (Char.valid n)) => Option.none Char)
					t)
				(Option.some Char (Char.intro n h)))
		((h2 : Char.valid n) => Eq.refl (Option Char) (Option.some Char (Char.intro n h)))
		((h2 : Prop.Not (Char.valid n)) =>
			False.elim
				(Eq (Option Char) (Option.none Char) (Option.some Char (Char.intro n h)))
				(h2 h))
		(Char.dec_valid n);

Char.try_of_nat_none : (n : Nat) -> Prop.Not (Char.valid n) ->
	Eq (Option Char) (Char.try_of_nat n) (Option.none Char) :=
	(n : Nat) => (hn : Prop.Not (Char.valid n)) =>
	Decidable.rec_prop (Char.valid n)
		((t : Decidable (Char.valid n)) =>
			Eq (Option Char)
				(Decidable.rec (Char.valid n)
					((s : Decidable (Char.valid n)) => Option Char)
					((h2 : Char.valid n) => Option.some Char (Char.intro n h2))
					((h2 : Prop.Not (Char.valid n)) => Option.none Char)
					t)
				(Option.none Char))
		((h2 : Char.valid n) =>
			False.elim
				(Eq (Option Char) (Option.some Char (Char.intro n h2)) (Option.none Char))
				(hn h2))
		((h2 : Prop.Not (Char.valid n)) => Eq.refl (Option Char) (Option.none Char))
		(Char.dec_valid n);

// U+FFFD, the replacement character.
Char.valid_replacement : Char.valid 65533 :=
	Prop.And.intro
		(Nat.Lt 65533 1114112)
		(Prop.Not (Prop.And (Nat.Le 55296 65533) (Nat.Le 65533 57343)))
		(Exists.intro Nat
			((k : Nat) => Eq Nat (Nat.add (Nat.succ 65533) k) 1114112)
			1048578
			(Eq.refl Nat 1114112))
		((h : Prop.And (Nat.Le 55296 65533) (Nat.Le 65533 57343)) =>
			Nat.lt_not_le 57343 65533
				(Exists.intro Nat
					((k : Nat) => Eq Nat (Nat.add (Nat.succ 57343) k) 65533)
					8189
					(Eq.refl Nat 65533))
				(Prop.And.right (Nat.Le 55296 65533) (Nat.Le 65533 57343) h));

Char.replacement : Char := Char.intro 65533 Char.valid_replacement;

// A character from any number: a valid code point is itself, anything else
// becomes the replacement character.
Char.of_nat : Nat -> Char :=
	(n : Nat) => Option.unwrap_or Char (Char.try_of_nat n) Char.replacement;

Char.of_nat_valid : (n : Nat) -> (h : Char.valid n) ->
	Eq Char (Char.of_nat n) (Char.intro n h) :=
	(n : Nat) => (h : Char.valid n) =>
	Eq.congr (Option Char) Char
		((o : Option Char) => Option.unwrap_or Char o Char.replacement)
		(Char.try_of_nat n) (Option.some Char (Char.intro n h))
		(Char.try_of_nat_some n h);

Char.of_nat_invalid : (n : Nat) -> Prop.Not (Char.valid n) ->
	Eq Char (Char.of_nat n) Char.replacement :=
	(n : Nat) => (hn : Prop.Not (Char.valid n)) =>
	Eq.congr (Option Char) Char
		((o : Option Char) => Option.unwrap_or Char o Char.replacement)
		(Char.try_of_nat n) (Option.none Char)
		(Char.try_of_nat_none n hn);

@lang(String)
String : Type := List Char;
@lang(String.nil)
String.nil : String := List.nil Char;
@lang(String.cons)
String.cons : Char -> String -> String := List.cons Char;
