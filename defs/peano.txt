@require "equality.txt"

@inductive
Nat : Type;
@intro(Nat)
Nat.zero : Nat;
@intro(Nat)
Nat.succ : Nat -> Nat;
@rec(Nat)
Nat.rec :
	(motive : Nat -> Type) ->
	(case_zero : motive Nat.zero) ->
	(case_succ : (n : Nat) -> motive n -> motive (Nat.succ n)) ->
	(target : Nat) ->
	motive target;

@iota(Nat)
Nat.rec motive case_zero case_succ Nat.zero : motive Nat.zero := case_zero;
@iota(Nat)
Nat.rec motive case_zero case_succ (Nat.succ n) : motive (Nat.succ n) :=
	case_succ n (Nat.rec motive case_zero case_succ n);

Nat.rec_prop :
	(motive : Nat -> Prop) ->
	(case_zero : motive Nat.zero) ->
	(case_succ : (n : Nat) -> motive n -> motive (Nat.succ n)) ->
	(target : Nat) ->
	motive target :=
	(motive : Nat -> Prop) =>
	(case_zero : motive Nat.zero) =>
	(case_succ : (n : Nat) -> motive n -> motive (Nat.succ n)) =>
	(target : Nat) =>
	Lift.elim
		(motive target)
		(Nat.rec
			((n : Nat) => Lift (motive n))
			(Lift.intro (motive Nat.zero) case_zero)
			((n : Nat) =>
				(x : Lift (motive n)) =>
				Lift.intro
					(motive (Nat.succ n))
					(case_succ n (Lift.elim (motive n) x)))
			target);

Nat.one : Nat := Nat.succ Nat.zero;
Nat.two : Nat := Nat.succ Nat.one;
	
// Addition, by recursion on the second argument.
Nat.Add : Nat -> Nat -> Nat := (a : Nat) => (b : Nat) =>
	Nat.rec
		((n : Nat) => Nat)
		a
		// a + succ(b) = succ(ih), ih = a + b
		((n : Nat) => (ih : Nat) => Nat.succ ih)
		b;

// pred 0 = 0, and a - b = 0 once b reaches a: truncated, as it has to be
// without negative numbers.
Nat.pred : Nat -> Nat :=
	(n : Nat) =>
	Nat.rec ((t : Nat) => Nat) Nat.zero ((m : Nat) => (ih : Nat) => m) n;

Nat.double : Nat -> Nat :=
	(n : Nat) =>
	Nat.rec
		((t : Nat) => Nat)
		Nat.zero
		((m : Nat) => (ih : Nat) => Nat.succ (Nat.succ ih))
		n;

Nat.Sub : Nat -> Nat -> Nat :=
	(a : Nat) => (b : Nat) =>
	Nat.rec ((t : Nat) => Nat) a ((m : Nat) => (ih : Nat) => Nat.pred ih) b;

// ∀ x ∈ ℕ. x + 0 = x
Nat.Add.add_zero : (x : Nat) -> Eq Nat (Nat.Add x Nat.zero) x :=
	(x : Nat) => Eq.refl Nat x;
// ∀ x, y ∈ ℕ. x + S y = S (x + y)
Nat.Add.add_succ : (x : Nat) -> (y : Nat) -> Eq Nat (Nat.Add x (Nat.succ y)) (Nat.succ (Nat.Add x y)) :=
	(x : Nat) => (y : Nat) => Eq.refl Nat (Nat.succ (Nat.Add x y));

// ∀ x ∈ ℕ. 0 + x = x
Nat.Add.zero_add : (x : Nat) -> Eq Nat (Nat.Add Nat.zero x) x :=
	Nat.rec_prop
		((x : Nat) => Eq Nat (Nat.Add Nat.zero x) x)
		(Nat.Add.add_zero Nat.zero) // 0 + 0 = 0
		(
			// goal : 0 + x = x -> 0 + S x = S x
			// h : 0 + x = x := intro on goal
			// l1 : 0 + S x = S (0 + x) := add_succ on 0 x
			// l2 : S (0 + x) = S x := Eq.congr on S h
			// 0 + S x = S x := Eq.trans on l1 l2
			(x : Nat) => (h : Eq Nat (Nat.Add Nat.zero x) x) =>
			Eq.trans Nat
				(Nat.Add Nat.zero (Nat.succ x)) // 0 + S x
				(Nat.succ (Nat.Add Nat.zero x)) // S (0 + x)
				(Nat.succ x) // S x
				(Nat.Add.add_succ Nat.zero x) // 0 + S x = S (0 + x)
				(Eq.congr Nat Nat Nat.succ (Nat.Add Nat.zero x) x h)
		);


// ∀ x ∈ ℕ. S x + y = S (x + y)
Nat.Add.succ_add : (x : Nat) -> (y : Nat) -> Eq Nat (Nat.Add (Nat.succ x) y) (Nat.succ (Nat.Add x y)) :=
	(x : Nat) => Nat.rec_prop
		((y : Nat) => Eq Nat (Nat.Add (Nat.succ x) y) (Nat.succ (Nat.Add x y)))
		// goal: S x + 0 = S (x + 0)
		// l1: S x + 0 = S x := add_zero on S x
		// l2: x + 0 = x := add_zero on x
		// l3: x = x + 0 := Eq.symm on l2
		// l4: S x = S (x + 0) := Eq.congr S on l3
		// S x + 0 = S (x + 0) := Eq.trans on l1 l4
		(
			Eq.trans Nat
				(Nat.Add (Nat.succ x) Nat.zero)
				(Nat.succ x)
				(Nat.succ (Nat.Add x Nat.zero))
				(Nat.Add.add_zero (Nat.succ x))
				(
					Eq.congr Nat Nat Nat.succ x (Nat.Add x Nat.zero)
						(Eq.symm Nat (Nat.Add x Nat.zero) x (Nat.Add.add_zero x))
				)
		)

		// goal: S x + y = S (x + y) -> S x + S y = S (x + S y)
		// h : S x + y = S (x + y) := intro on goal
		// l1: x + S y = S (x + y) := add_succ on x y
		// l2: S (x + y) = x + S y := Eq.symm on l1
		// l3: S x + y = x + S y := Eq.trans on h, l2
		// l4: S (S x + y) = S (x + S y) := Eq.congr S on l3
		// l5: S x + S y = S (S x + y) := add_succ on (S x) y
		// goal: S x + S y = S (x + S y) := Eq.trans on l5 l4
		// 
		(
			(y : Nat) => (h : Eq Nat (Nat.Add (Nat.succ x) y) (Nat.succ (Nat.Add x y))) =>
			Eq.trans Nat
				(Nat.Add (Nat.succ x) (Nat.succ y))
				(Nat.succ (Nat.Add (Nat.succ x) y))
				(Nat.succ (Nat.Add x (Nat.succ y)))
				(Nat.Add.add_succ (Nat.succ x) y)
				(Eq.congr Nat Nat Nat.succ
					(Nat.Add (Nat.succ x) y)
					(Nat.Add x (Nat.succ y))
					(Eq.trans Nat
						(Nat.Add (Nat.succ x) y)
						(Nat.succ (Nat.Add x y))
						(Nat.Add x (Nat.succ y))
						h
						(Eq.symm Nat
							(Nat.Add x (Nat.succ y))
							(Nat.succ (Nat.Add x y))
							(Nat.Add.add_succ x y)
						)
					)
				)
		);

// succ is injective: pred undoes it, and pred computes.
Nat.succ_inj : (a : Nat) -> (b : Nat) ->
	Eq Nat (Nat.succ a) (Nat.succ b) -> Eq Nat a b :=
	(a : Nat) => (b : Nat) => (h : Eq Nat (Nat.succ a) (Nat.succ b)) =>
	Eq.congr Nat Nat Nat.pred (Nat.succ a) (Nat.succ b) h;

// True at zero and False at every successor, so transporting True.triv along a
// supposed equality between the two lands on False.
Nat.is_zero : Nat -> Prop :=
	(n : Nat) =>
	Nat.rec ((t : Nat) => Prop) True ((m : Nat) => (ih : Prop) => False) n;

Nat.zero_ne_succ : (n : Nat) -> Eq Nat Nat.zero (Nat.succ n) -> False :=
	(n : Nat) => (h : Eq Nat Nat.zero (Nat.succ n)) =>
	Eq.rec_prop Nat Nat.zero
		((b : Nat) => (p : Eq Nat Nat.zero b) => Nat.is_zero b)
		True.triv
		(Nat.succ n)
		h;

Nat.succ_ne_zero : (n : Nat) -> Eq Nat (Nat.succ n) Nat.zero -> False :=
	(n : Nat) => (h : Eq Nat (Nat.succ n) Nat.zero) =>
	Nat.zero_ne_succ n (Eq.symm Nat (Nat.succ n) Nat.zero h);

// double is injective. Induction on the first argument with the second still
// quantified, then cases on the second.
Nat.double_inj.zero : (b : Nat) ->
	Eq Nat (Nat.double Nat.zero) (Nat.double b) -> Eq Nat Nat.zero b :=
	Nat.rec_prop
		((b : Nat) =>
			Eq Nat (Nat.double Nat.zero) (Nat.double b) -> Eq Nat Nat.zero b)
		((h : Eq Nat Nat.zero Nat.zero) => Eq.refl Nat Nat.zero)
		((b : Nat) =>
			(ih : Eq Nat (Nat.double Nat.zero) (Nat.double b) ->
				Eq Nat Nat.zero b) =>
			(h : Eq Nat Nat.zero (Nat.succ (Nat.succ (Nat.double b)))) =>
			False.elim (Eq Nat Nat.zero (Nat.succ b))
				(Nat.zero_ne_succ (Nat.succ (Nat.double b)) h));

Nat.double_inj.succ : (a : Nat) ->
	((b : Nat) -> Eq Nat (Nat.double a) (Nat.double b) -> Eq Nat a b) ->
	(b : Nat) ->
	Eq Nat (Nat.double (Nat.succ a)) (Nat.double b) -> Eq Nat (Nat.succ a) b :=
	(a : Nat) =>
	(ih : (b : Nat) -> Eq Nat (Nat.double a) (Nat.double b) -> Eq Nat a b) =>
	Nat.rec_prop
		((b : Nat) =>
			Eq Nat (Nat.double (Nat.succ a)) (Nat.double b) ->
			Eq Nat (Nat.succ a) b)
		((h : Eq Nat (Nat.succ (Nat.succ (Nat.double a))) Nat.zero) =>
			False.elim (Eq Nat (Nat.succ a) Nat.zero)
				(Nat.succ_ne_zero (Nat.succ (Nat.double a)) h))
		((b : Nat) =>
			(ihb : Eq Nat (Nat.double (Nat.succ a)) (Nat.double b) ->
				Eq Nat (Nat.succ a) b) =>
			(h : Eq Nat
				(Nat.succ (Nat.succ (Nat.double a)))
				(Nat.succ (Nat.succ (Nat.double b)))) =>
			Eq.congr Nat Nat Nat.succ a b
				(ih b
					(Nat.succ_inj (Nat.double a) (Nat.double b)
						(Nat.succ_inj
							(Nat.succ (Nat.double a))
							(Nat.succ (Nat.double b))
							h))));

Nat.double_inj : (a : Nat) -> (b : Nat) ->
	Eq Nat (Nat.double a) (Nat.double b) -> Eq Nat a b :=
	Nat.rec_prop
		((a : Nat) =>
			(b : Nat) -> Eq Nat (Nat.double a) (Nat.double b) -> Eq Nat a b)
		Nat.double_inj.zero
		Nat.double_inj.succ;

Nat.Mul : (a : Nat) -> (b : Nat) -> Nat := (a : Nat) => (b : Nat) =>
	Nat.rec
		((n : Nat) => Nat)
		// case_zero: a * 0 = 0
		Nat.zero
		// case_succ: a * succ(b) = a + ih, ih = a * b
		((n : Nat) => (ih : Nat) => Nat.Add a ih)
		b;

// forall a : Nat. a * 0 = 0
Nat.Mul.mul_zero : (a : Nat) -> Eq Nat (Nat.Mul a Nat.zero) Nat.zero :=
	(a : Nat) => Eq.refl Nat Nat.zero;

// forall a : Nat. a * succ(b) = a + a * b
Nat.Mul.mul_succ : (a : Nat) -> (b : Nat) -> Eq Nat (Nat.Mul a (Nat.succ b)) (Nat.Add a (Nat.Mul a b)) :=
	(a : Nat) => (b : Nat) => Eq.refl Nat (Nat.Add a (Nat.Mul a b));
