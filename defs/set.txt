@require "prop.txt"
@require "equality.txt"
@require "peano.txt"

Set : (A : Type) -> Type := (A : Type) => A -> Prop;

// The set every element is in. `Sub A (Set.Univ A)` is a copy of A.
Set.Univ : (A : Type) -> Set A := (A : Type) => (a : A) => True;

Injective : (A : Type) -> (B : Type) -> (f : A -> B) -> Prop :=
	(A : Type) => (B : Type) => (f : A -> B) =>
	(a : A) -> (b : A) -> Eq B (f a) (f b) -> Eq A a b;

// f reaches every b in B.
Surjective : (A : Type) -> (B : Type) -> (f : A -> B) -> Prop :=
	(A : Type) => (B : Type) => (f : A -> B) =>
	(b : B) -> Exists A ((a : A) => Eq B (f a) b);

Bijective : (A : Type) -> (B : Type) -> (f : A -> B) -> Prop :=
	(A : Type) => (B : Type) => (f : A -> B) =>
	Prop.And (Injective A B f) (Surjective A B f);

Bijective.intro : (A : Type) -> (B : Type) -> (f : A -> B) ->
	Injective A B f -> Surjective A B f -> Bijective A B f :=
	(A : Type) => (B : Type) => (f : A -> B) =>
	(inj : Injective A B f) => (sur : Surjective A B f) =>
	Prop.And.intro (Injective A B f) (Surjective A B f) inj sur;

Bijection : (A : Type) -> (B : Type) -> Prop :=
	(A : Type) => (B : Type) =>
	Exists (A -> B) ((f : A -> B) => Bijective A B f);

Bijection.intro : (A : Type) -> (B : Type) -> (f : A -> B) ->
	Bijective A B f -> Bijection A B :=
	(A : Type) => (B : Type) => (f : A -> B) => (bij : Bijective A B f) =>
	Exists.intro (A -> B) ((g : A -> B) => Bijective A B g) f bij;

Bijection.refl : (A : Type) -> Bijection A A :=
	(A : Type) =>
	Bijection.intro A A ((a : A) => a)
		(Bijective.intro A A ((a : A) => a)
			((a : A) => (b : A) => (h : Eq A a b) => h)
			((b : A) =>
				Exists.intro A ((a : A) => Eq A a b) b (Eq.refl A b)));

// ---------------------------------------------------------------------------
// Quotients
//
// The domain is a Type and so is the quotient: a quotient that landed in a
// smaller universe than the type it quotients would put a quotient of Type
// inside Type, which is where paradoxes come from. Nothing here is universe
// polymorphic, so this is the quotient of small types only - enough for the
// integers or the rationals, not for Type itself.
//
// Quot.lift has no computation rule: `Quot.lift ... (Quot.mk ... a)` does not
// reduce to `f a`, because a rule cannot repeat a pattern variable the way
// matching on Quot.mk A R a under Quot.lift A R would need to. Quot.eq is the
// only one of these with any force today.
// ---------------------------------------------------------------------------

@axiom
Quot : (A : Type) -> (A -> A -> Prop) -> Type;

@axiom
Quot.mk : (A : Type) -> (R : A -> A -> Prop) -> A -> Quot A R;

@axiom
Quot.eq : (A : Type) -> (R : A -> A -> Prop) ->
    (a : A) -> (b : A) -> R a b ->
    Eq (Quot A R) (Quot.mk A R a) (Quot.mk A R b);

@axiom
Quot.lift :
    (A : Type) -> (R : A -> A -> Prop) ->
    (B : Type) ->
    (f : A -> B) ->
    ((a : A) -> (b : A) -> R a b -> Eq B (f a) (f b)) ->
    Quot A R -> B;

@inductive
Sub : (A : Type) -> (s : Set A) -> Type;
@intro(Sub)
Sub.intro : (A : Type) -> (s : Set A) -> (a : A) -> s a -> Sub A s;
@rec(Sub)
Sub.rec :
	(A : Type) -> (s : Set A) ->
	(motive : Sub A s -> Type) ->
	(case_intro : (a : A) -> (h : s a) -> motive (Sub.intro A s a h)) ->
	(target : Sub A s) ->
	motive target;

@iota(Sub)
Sub.rec A s motive case_intro (Sub.intro A s a h) :
	motive (Sub.intro A s a h) := case_intro a h;

Sub.val : (A : Type) -> (s : Set A) -> Sub A s -> A :=
	(A : Type) => (s : Set A) => (x : Sub A s) =>
	Sub.rec A s
		((t : Sub A s) => A)
		((a : A) => (h : s a) => a)
		x;

Sub.rec_prop :
	(A : Type) -> (s : Set A) ->
	(motive : Sub A s -> Prop) ->
	(case_intro : (a : A) -> (h : s a) -> motive (Sub.intro A s a h)) ->
	(target : Sub A s) ->
	motive target :=
	(A : Type) => (s : Set A) =>
	(motive : Sub A s -> Prop) =>
	(case_intro : (a : A) -> (h : s a) -> motive (Sub.intro A s a h)) =>
	(target : Sub A s) =>
	Lift.elim
		(motive target)
		(Sub.rec A s
			((x : Sub A s) => Lift (motive x))
			((a : A) => (h : s a) =>
				Lift.intro (motive (Sub.intro A s a h)) (case_intro a h))
			target);

// `Sub.val A s (Sub.intro A s a h)` reduces to `a`, so the proof the
// constructor carries is exactly the proof required.
Sub.property : (A : Type) -> (s : Set A) -> (x : Sub A s) ->
	s (Sub.val A s x) :=
	(A : Type) => (s : Set A) => (x : Sub A s) =>
	Sub.rec_prop A s
		((t : Sub A s) => s (Sub.val A s t))
		((a : A) => (h : s a) => h)
		x;

// Two elements of Sub A s that hold the same value differ only in which proof
// of membership they carry, and proofs of a Prop are interchangeable.
Sub.intro_eq : (A : Type) -> (s : Set A) -> (a : A) -> (ha : s a) ->
	(b : A) -> (hb : s b) -> Eq A a b ->
	Eq (Sub A s) (Sub.intro A s a ha) (Sub.intro A s b hb) :=
	(A : Type) => (s : Set A) => (a : A) => (ha : s a) =>
	(b : A) => (hb : s b) => (e : Eq A a b) =>
	Eq.rec_prop A a
		((c : A) => (p : Eq A a c) =>
			(hc : s c) -> Eq (Sub A s) (Sub.intro A s a ha) (Sub.intro A s c hc))
		((hc : s a) => Eq.refl (Sub A s) (Sub.intro A s a ha))
		b
		e
		hb;

// so the value determines the element
Sub.val_inj : (A : Type) -> (s : Set A) -> (x : Sub A s) -> (y : Sub A s) ->
	Eq A (Sub.val A s x) (Sub.val A s y) -> Eq (Sub A s) x y :=
	(A : Type) => (s : Set A) => (x : Sub A s) =>
	Sub.rec_prop A s
		((t : Sub A s) => (y : Sub A s) ->
			Eq A (Sub.val A s t) (Sub.val A s y) -> Eq (Sub A s) t y)
		((a : A) => (ha : s a) => (y : Sub A s) =>
			Sub.rec_prop A s
				((u : Sub A s) =>
					Eq A (Sub.val A s (Sub.intro A s a ha)) (Sub.val A s u) ->
					Eq (Sub A s) (Sub.intro A s a ha) u)
				((b : A) => (hb : s b) =>
					(e : Eq A (Sub.val A s (Sub.intro A s a ha))
						(Sub.val A s (Sub.intro A s b hb))) =>
					Sub.intro_eq A s a ha b hb e)
				y)
		x;

Set.Card.Eq : (A : Type) -> Set A -> Set A -> Prop :=
	(A : Type) => (s : Set A) => (t : Set A) =>
	Bijection (Sub A s) (Sub A t);

Set.Card.Eq.refl : (A : Type) -> (s : Set A) -> Set.Card.Eq A s s :=
	(A : Type) => (s : Set A) => Bijection.refl (Sub A s);
