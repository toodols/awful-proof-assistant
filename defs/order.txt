@require "double.txt"
@require "bool.txt"

Nat.Le : Nat -> Nat -> Prop :=
	(m : Nat) => (n : Nat) => Exists Nat ((k : Nat) => Eq Nat (Nat.Add m k) n);

Nat.Lt : Nat -> Nat -> Prop :=
	(m : Nat) => (n : Nat) => Nat.Le (Nat.succ m) n;

Nat.lt_succ : (n : Nat) -> Nat.Lt n (Nat.succ n) :=
	(n : Nat) =>
	Exists.intro Nat
		((k : Nat) => Eq Nat (Nat.Add (Nat.succ n) k) (Nat.succ n))
		Nat.zero
		(Eq.refl Nat (Nat.succ n));

Nat.not_lt_zero : (m : Nat) -> Nat.Lt m Nat.zero -> False :=
	(m : Nat) => (h : Nat.Lt m Nat.zero) =>
	Exists.elim Nat ((j : Nat) => Eq Nat (Nat.Add (Nat.succ m) j) Nat.zero)
		False h
		((j : Nat) => (e : Eq Nat (Nat.Add (Nat.succ m) j) Nat.zero) =>
			Nat.succ_ne_zero (Nat.Add m j)
				(Eq.trans Nat
					(Nat.succ (Nat.Add m j))
					(Nat.Add (Nat.succ m) j)
					Nat.zero
					(Eq.symm Nat (Nat.Add (Nat.succ m) j) (Nat.succ (Nat.Add m j))
						(Nat.Add.succ_add m j))
					e));

Nat.strong_ind : (P : Nat -> Prop) ->
	((n : Nat) -> ((m : Nat) -> Nat.Lt m n -> P m) -> P n) ->
	(n : Nat) -> P n :=
	(P : Nat -> Prop) =>
	(step : (n : Nat) -> ((m : Nat) -> Nat.Lt m n -> P m) -> P n) =>
	(n : Nat) =>
	Nat.rec_prop
		((t : Nat) => (m : Nat) -> Nat.Lt m t -> P m)
		((m : Nat) => (h : Nat.Lt m Nat.zero) =>
			False.elim (P m) (Nat.not_lt_zero m h))
		((t : Nat) =>
			(ih : (m : Nat) -> Nat.Lt m t -> P m) =>
			(m : Nat) => (h : Nat.Lt m (Nat.succ t)) =>
			Exists.elim Nat
				((j : Nat) => Eq Nat (Nat.Add (Nat.succ m) j) (Nat.succ t))
				(P m) h
				((j : Nat) => (e : Eq Nat (Nat.Add (Nat.succ m) j) (Nat.succ t)) =>
					Nat.cases
						((i : Nat) => Eq Nat (Nat.Add m i) t -> P m)
						((e2 : Eq Nat m t) =>
							Eq.subst Nat t m P (Eq.symm Nat m t e2) (step t ih))
						((i : Nat) => (e2 : Eq Nat (Nat.succ (Nat.Add m i)) t) =>
							ih m
								(Exists.intro Nat
									((r : Nat) => Eq Nat (Nat.Add (Nat.succ m) r) t)
									i
									(Eq.trans Nat
										(Nat.Add (Nat.succ m) i)
										(Nat.succ (Nat.Add m i))
										t
										(Nat.Add.succ_add m i)
										e2)))
						j
						(Nat.succ_inj (Nat.Add m j) t
							(Eq.trans Nat
								(Nat.succ (Nat.Add m j))
								(Nat.Add (Nat.succ m) j)
								(Nat.succ t)
								(Eq.symm Nat (Nat.Add (Nat.succ m) j) (Nat.succ (Nat.Add m j))
									(Nat.Add.succ_add m j))
								e))))
		(Nat.succ n)
		n
		(Nat.lt_succ n);

Nat.le_zero : (p : Nat) -> Nat.Le p Nat.zero -> Eq Nat p Nat.zero :=
	(p : Nat) => (h : Nat.Le p Nat.zero) =>
	Exists.elim Nat ((j : Nat) => Eq Nat (Nat.Add p j) Nat.zero)
		(Eq Nat p Nat.zero) h
		((j : Nat) => (e : Eq Nat (Nat.Add p j) Nat.zero) =>
			Nat.cases
				((q : Nat) => Eq Nat (Nat.Add q j) Nat.zero -> Eq Nat q Nat.zero)
				((e0 : Eq Nat (Nat.Add Nat.zero j) Nat.zero) => Eq.refl Nat Nat.zero)
				((q : Nat) => (e1 : Eq Nat (Nat.Add (Nat.succ q) j) Nat.zero) =>
					False.elim (Eq Nat (Nat.succ q) Nat.zero)
						(Nat.succ_ne_zero (Nat.Add q j)
							(Eq.trans Nat
								(Nat.succ (Nat.Add q j))
								(Nat.Add (Nat.succ q) j)
								Nat.zero
								(Eq.symm Nat (Nat.Add (Nat.succ q) j) (Nat.succ (Nat.Add q j))
									(Nat.Add.succ_add q j))
								e1)))
				p
				e);

Nat.one_lt_form : (k : Nat) -> Nat.Lt (Nat.succ Nat.zero) k ->
	Exists Nat ((i : Nat) => Eq Nat k (Nat.succ (Nat.succ i))) :=
	(k : Nat) => (h : Nat.Lt (Nat.succ Nat.zero) k) =>
	Exists.elim Nat ((i : Nat) => Eq Nat (Nat.Add Nat.two i) k)
		(Exists Nat ((i : Nat) => Eq Nat k (Nat.succ (Nat.succ i))))
		h
		((i : Nat) => (e : Eq Nat (Nat.Add Nat.two i) k) =>
			Exists.intro Nat ((j : Nat) => Eq Nat k (Nat.succ (Nat.succ j))) i
				(Eq.trans Nat k (Nat.Add Nat.two i) (Nat.succ (Nat.succ i))
					(Eq.symm Nat (Nat.Add Nat.two i) k e)
					(Nat.Add.two_add i)));

Nat.one_lt_ne_zero : (k : Nat) -> Nat.Lt (Nat.succ Nat.zero) k ->
	NotEq Nat k Nat.zero :=
	(k : Nat) => (h : Nat.Lt (Nat.succ Nat.zero) k) =>
	Exists.elim Nat ((i : Nat) => Eq Nat k (Nat.succ (Nat.succ i)))
		(NotEq Nat k Nat.zero)
		(Nat.one_lt_form k h)
		((i : Nat) => (e : Eq Nat k (Nat.succ (Nat.succ i))) =>
			(z : Eq Nat k Nat.zero) =>
			Nat.succ_ne_zero (Nat.succ i)
				(Eq.trans Nat
					(Nat.succ (Nat.succ i))
					k
					Nat.zero
					(Eq.symm Nat k (Nat.succ (Nat.succ i)) e)
					z));

Nat.one_lt_cases : (x : Nat) ->
	NotEq Nat x Nat.zero ->
	NotEq Nat x (Nat.succ Nat.zero) ->
	Nat.Lt (Nat.succ Nat.zero) x :=
	Nat.cases
		((x : Nat) =>
			NotEq Nat x Nat.zero ->
			NotEq Nat x (Nat.succ Nat.zero) ->
			Nat.Lt (Nat.succ Nat.zero) x)
		((h0 : NotEq Nat Nat.zero Nat.zero) =>
			(h1 : NotEq Nat Nat.zero (Nat.succ Nat.zero)) =>
			False.elim (Nat.Lt (Nat.succ Nat.zero) Nat.zero)
				(h0 (Eq.refl Nat Nat.zero)))
		((n : Nat) =>
			Nat.cases
				((m : Nat) =>
					NotEq Nat (Nat.succ m) Nat.zero ->
					NotEq Nat (Nat.succ m) (Nat.succ Nat.zero) ->
					Nat.Lt (Nat.succ Nat.zero) (Nat.succ m))
				((h0 : NotEq Nat (Nat.succ Nat.zero) Nat.zero) =>
					(h1 : NotEq Nat (Nat.succ Nat.zero) (Nat.succ Nat.zero)) =>
					False.elim (Nat.Lt (Nat.succ Nat.zero) (Nat.succ Nat.zero))
						(h1 (Eq.refl Nat (Nat.succ Nat.zero))))
				((m : Nat) =>
					(h0 : NotEq Nat (Nat.succ (Nat.succ m)) Nat.zero) =>
					(h1 : NotEq Nat (Nat.succ (Nat.succ m)) (Nat.succ Nat.zero)) =>
					Exists.intro Nat
						((j : Nat) =>
							Eq Nat (Nat.Add Nat.two j) (Nat.succ (Nat.succ m)))
						m
						(Nat.Add.two_add m))
				n);

Nat.le_lt_total : (x : Nat) -> (y : Nat) -> Prop.Or (Nat.Le x y) (Nat.Lt y x) :=
	Nat.rec_prop
		((x : Nat) => (y : Nat) -> Prop.Or (Nat.Le x y) (Nat.Lt y x))
		((y : Nat) =>
			Prop.Or.inl (Nat.Le Nat.zero y) (Nat.Lt y Nat.zero)
				(Exists.intro Nat ((t : Nat) => Eq Nat (Nat.Add Nat.zero t) y) y
					(Nat.Add.zero_add y)))
		((x : Nat) =>
			(ih : (y : Nat) -> Prop.Or (Nat.Le x y) (Nat.Lt y x)) =>
			(y : Nat) =>
			Prop.Or.elim (Nat.Le x y) (Nat.Lt y x)
				(Prop.Or (Nat.Le (Nat.succ x) y) (Nat.Lt y (Nat.succ x)))
				(ih y)
				((h : Nat.Le x y) =>
					Exists.elim Nat ((t : Nat) => Eq Nat (Nat.Add x t) y)
						(Prop.Or (Nat.Le (Nat.succ x) y) (Nat.Lt y (Nat.succ x)))
						h
						((t : Nat) => (e : Eq Nat (Nat.Add x t) y) =>
							Nat.cases
								((s : Nat) =>
									Eq Nat (Nat.Add x s) y ->
									Prop.Or (Nat.Le (Nat.succ x) y) (Nat.Lt y (Nat.succ x)))
								((e0 : Eq Nat x y) =>
									Prop.Or.inr (Nat.Le (Nat.succ x) y) (Nat.Lt y (Nat.succ x))
										(Exists.intro Nat
											((j : Nat) => Eq Nat (Nat.Add (Nat.succ y) j) (Nat.succ x))
											Nat.zero
											(Eq.congr Nat Nat Nat.succ y x (Eq.symm Nat x y e0))))
								((s : Nat) => (e1 : Eq Nat (Nat.succ (Nat.Add x s)) y) =>
									Prop.Or.inl (Nat.Le (Nat.succ x) y) (Nat.Lt y (Nat.succ x))
										(Exists.intro Nat
											((j : Nat) => Eq Nat (Nat.Add (Nat.succ x) j) y)
											s
											(Eq.trans Nat
												(Nat.Add (Nat.succ x) s)
												(Nat.succ (Nat.Add x s))
												y
												(Nat.Add.succ_add x s)
												e1)))
								t
								e))
				((h : Nat.Lt y x) =>
					Exists.elim Nat ((j : Nat) => Eq Nat (Nat.Add (Nat.succ y) j) x)
						(Prop.Or (Nat.Le (Nat.succ x) y) (Nat.Lt y (Nat.succ x)))
						h
						((j : Nat) => (e : Eq Nat (Nat.Add (Nat.succ y) j) x) =>
							Prop.Or.inr (Nat.Le (Nat.succ x) y) (Nat.Lt y (Nat.succ x))
								(Exists.intro Nat
									((r : Nat) => Eq Nat (Nat.Add (Nat.succ y) r) (Nat.succ x))
									(Nat.succ j)
									(Eq.congr Nat Nat Nat.succ
										(Nat.Add (Nat.succ y) j) x e)))));


// ---- no number is below itself ----

Nat.add_succ_ne_self : (b : Nat) -> (j : Nat) ->
	NotEq Nat (Nat.Add (Nat.succ b) j) b :=
	Nat.rec_prop
		((b : Nat) => (j : Nat) -> NotEq Nat (Nat.Add (Nat.succ b) j) b)
		((j : Nat) => (e : Eq Nat (Nat.Add (Nat.succ Nat.zero) j) Nat.zero) =>
			Nat.succ_ne_zero (Nat.Add Nat.zero j)
				(Eq.trans Nat
					(Nat.succ (Nat.Add Nat.zero j))
					(Nat.Add (Nat.succ Nat.zero) j)
					Nat.zero
					(Eq.symm Nat (Nat.Add (Nat.succ Nat.zero) j)
						(Nat.succ (Nat.Add Nat.zero j))
						(Nat.Add.succ_add Nat.zero j))
					e))
		((b : Nat) =>
			(ih : (j : Nat) -> NotEq Nat (Nat.Add (Nat.succ b) j) b) =>
			(j : Nat) => (e : Eq Nat (Nat.Add (Nat.succ (Nat.succ b)) j) (Nat.succ b)) =>
			ih j
				(Nat.succ_inj (Nat.Add (Nat.succ b) j) b
					(Eq.trans Nat
						(Nat.succ (Nat.Add (Nat.succ b) j))
						(Nat.Add (Nat.succ (Nat.succ b)) j)
						(Nat.succ b)
						(Eq.symm Nat (Nat.Add (Nat.succ (Nat.succ b)) j)
							(Nat.succ (Nat.Add (Nat.succ b) j))
							(Nat.Add.succ_add (Nat.succ b) j))
						e)));

Nat.lt_irrefl : (b : Nat) -> Nat.Lt b b -> False :=
	(b : Nat) => (h : Nat.Lt b b) =>
	Exists.elim Nat ((j : Nat) => Eq Nat (Nat.Add (Nat.succ b) j) b) False h
		((j : Nat) => (e : Eq Nat (Nat.Add (Nat.succ b) j) b) =>
			Nat.add_succ_ne_self b j e);

Nat.le_trans : (a : Nat) -> (b : Nat) -> (c : Nat) ->
	Nat.Le a b -> Nat.Le b c -> Nat.Le a c :=
	(a : Nat) => (b : Nat) => (c : Nat) =>
	(h1 : Nat.Le a b) => (h2 : Nat.Le b c) =>
	Exists.elim Nat ((k : Nat) => Eq Nat (Nat.Add a k) b) (Nat.Le a c) h1
		((k : Nat) => (e1 : Eq Nat (Nat.Add a k) b) =>
			Exists.elim Nat ((j : Nat) => Eq Nat (Nat.Add b j) c) (Nat.Le a c) h2
				((j : Nat) => (e2 : Eq Nat (Nat.Add b j) c) =>
					Exists.intro Nat ((t : Nat) => Eq Nat (Nat.Add a t) c)
						(Nat.Add k j)
						(Eq.trans Nat
							(Nat.Add a (Nat.Add k j))
							(Nat.Add b j)
							c
							(Eq.trans Nat
								(Nat.Add a (Nat.Add k j))
								(Nat.Add (Nat.Add a k) j)
								(Nat.Add b j)
								(Eq.symm Nat (Nat.Add (Nat.Add a k) j) (Nat.Add a (Nat.Add k j))
									(Nat.Add.add_assoc a k j))
								(Eq.congr Nat Nat ((x : Nat) => Nat.Add x j) (Nat.Add a k) b e1))
							e2)));

// b < a and a <= b cannot both hold
Nat.lt_not_le : (b : Nat) -> (a : Nat) -> Nat.Lt b a -> Nat.Le a b -> False :=
	(b : Nat) => (a : Nat) => (h1 : Nat.Lt b a) => (h2 : Nat.Le a b) =>
	Nat.lt_irrefl b (Nat.le_trans (Nat.succ b) a b h1 h2);


// ---- deciding <= by computation ----

Nat.zero_le : (b : Nat) -> Nat.Le Nat.zero b :=
	(b : Nat) =>
	Exists.intro Nat ((t : Nat) => Eq Nat (Nat.Add Nat.zero t) b) b (Nat.Add.zero_add b);

Nat.le_succ_succ : (a : Nat) -> (b : Nat) -> Nat.Le a b -> Nat.Le (Nat.succ a) (Nat.succ b) :=
	(a : Nat) => (b : Nat) => (h : Nat.Le a b) =>
	Exists.elim Nat ((k : Nat) => Eq Nat (Nat.Add a k) b) (Nat.Le (Nat.succ a) (Nat.succ b)) h
		((k : Nat) => (e : Eq Nat (Nat.Add a k) b) =>
			Exists.intro Nat ((t : Nat) => Eq Nat (Nat.Add (Nat.succ a) t) (Nat.succ b)) k
				(Eq.trans Nat
					(Nat.Add (Nat.succ a) k)
					(Nat.succ (Nat.Add a k))
					(Nat.succ b)
					(Nat.Add.succ_add a k)
					(Eq.congr Nat Nat Nat.succ (Nat.Add a k) b e)));

Nat.le_of_succ_succ : (a : Nat) -> (b : Nat) -> Nat.Le (Nat.succ a) (Nat.succ b) -> Nat.Le a b :=
	(a : Nat) => (b : Nat) => (h : Nat.Le (Nat.succ a) (Nat.succ b)) =>
	Exists.elim Nat ((k : Nat) => Eq Nat (Nat.Add (Nat.succ a) k) (Nat.succ b)) (Nat.Le a b) h
		((k : Nat) => (e : Eq Nat (Nat.Add (Nat.succ a) k) (Nat.succ b)) =>
			Exists.intro Nat ((t : Nat) => Eq Nat (Nat.Add a t) b) k
				(Nat.succ_inj (Nat.Add a k) b
					(Eq.trans Nat
						(Nat.succ (Nat.Add a k))
						(Nat.Add (Nat.succ a) k)
						(Nat.succ b)
						(Eq.symm Nat (Nat.Add (Nat.succ a) k) (Nat.succ (Nat.Add a k))
							(Nat.Add.succ_add a k))
						e)));

// a <= b as a Bool, recursing on both numbers at once: the recursive call is
// the whole result, so it takes min(a, b) steps and no stack.
@lang(Nat.ble)
Nat.ble : Nat -> Nat -> Bool :=
	Nat.rec ((a : Nat) => Nat -> Bool)
		((b : Nat) => Bool.true)
		((a : Nat) => (ih : Nat -> Bool) => (b : Nat) =>
			Nat.rec ((t : Nat) => Bool) Bool.false ((c : Nat) => (r : Bool) => ih c) b);

Nat.le_of_ble : (a : Nat) -> (b : Nat) -> Eq Bool (Nat.ble a b) Bool.true -> Nat.Le a b :=
	Nat.rec_prop
		((a : Nat) => (b : Nat) -> Eq Bool (Nat.ble a b) Bool.true -> Nat.Le a b)
		((b : Nat) => (e : Eq Bool Bool.true Bool.true) => Nat.zero_le b)
		((a : Nat) =>
			(ih : (b : Nat) -> Eq Bool (Nat.ble a b) Bool.true -> Nat.Le a b) =>
			(b : Nat) =>
			Nat.cases
				((t : Nat) => Eq Bool (Nat.ble (Nat.succ a) t) Bool.true -> Nat.Le (Nat.succ a) t)
				((e : Eq Bool Bool.false Bool.true) =>
					False.elim (Nat.Le (Nat.succ a) Nat.zero) (Bool.false_ne_true e))
				((c : Nat) => (e : Eq Bool (Nat.ble a c) Bool.true) =>
					Nat.le_succ_succ a c (ih c e))
				b);

Nat.not_le_of_ble : (a : Nat) -> (b : Nat) -> Eq Bool (Nat.ble a b) Bool.false ->
	Prop.Not (Nat.Le a b) :=
	Nat.rec_prop
		((a : Nat) => (b : Nat) -> Eq Bool (Nat.ble a b) Bool.false -> Prop.Not (Nat.Le a b))
		((b : Nat) => (e : Eq Bool Bool.true Bool.false) =>
			False.elim (Prop.Not (Nat.Le Nat.zero b)) (Bool.true_ne_false e))
		((a : Nat) =>
			(ih : (b : Nat) -> Eq Bool (Nat.ble a b) Bool.false -> Prop.Not (Nat.Le a b)) =>
			(b : Nat) =>
			Nat.cases
				((t : Nat) =>
					Eq Bool (Nat.ble (Nat.succ a) t) Bool.false -> Prop.Not (Nat.Le (Nat.succ a) t))
				((e : Eq Bool Bool.false Bool.false) => (h : Nat.Le (Nat.succ a) Nat.zero) =>
					Nat.succ_ne_zero a (Nat.le_zero (Nat.succ a) h))
				((c : Nat) => (e : Eq Bool (Nat.ble a c) Bool.false) =>
					(h : Nat.Le (Nat.succ a) (Nat.succ c)) =>
					ih c e (Nat.le_of_succ_succ a c h))
				b);

// Both are computed from the same Bool, so evaluating this evaluates Nat.ble.
Nat.dec_le : (a : Nat) -> (b : Nat) -> Decidable (Nat.Le a b) :=
	(a : Nat) => (b : Nat) =>
	Bool.rec
		((t : Bool) => Eq Bool (Nat.ble a b) t -> Decidable (Nat.Le a b))
		((e : Eq Bool (Nat.ble a b) Bool.true) =>
			Decidable.yes (Nat.Le a b) (Nat.le_of_ble a b e))
		((e : Eq Bool (Nat.ble a b) Bool.false) =>
			Decidable.no (Nat.Le a b) (Nat.not_le_of_ble a b e))
		(Nat.ble a b)
		(Eq.refl Bool (Nat.ble a b));
