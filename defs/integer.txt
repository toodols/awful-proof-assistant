@require "arith.txt"

@inductive
Int : Type;
@intro(Int)
Int.nat : Nat -> Int;
@intro(Int)
Int.neg_succ : Nat -> Int;
@rec(Int)
Int.rec :
    (motive : Int -> Type) ->
    (case_nat : (n : Nat) -> motive (Int.nat n)) ->
    (case_neg_succ : (n : Nat) -> motive (Int.neg_succ n)) ->
    (target : Int) ->
    motive target;

@iota(Int)
Int.rec motive case_nat case_neg_succ (Int.nat n) :
	motive (Int.nat n) := case_nat n;
@iota(Int)
Int.rec motive case_nat case_neg_succ (Int.neg_succ n) :
	motive (Int.neg_succ n) := case_neg_succ n;

Int.rec_prop :
    (motive : Int -> Prop) ->
    (case_nat : (n : Nat) -> motive (Int.nat n)) ->
    (case_neg_succ : (n : Nat) -> motive (Int.neg_succ n)) ->
    (target : Int) ->
    motive target :=
    (motive : Int -> Prop) =>
    (case_nat : (n : Nat) -> motive (Int.nat n)) =>
    (case_neg_succ : (n : Nat) -> motive (Int.neg_succ n)) =>
    (target : Int) =>
    Lift.elim (motive target)
      (Int.rec
        ((x : Int) => Lift (motive x))
        ((n : Nat) => Lift.intro (motive (Int.nat n)) (case_nat n))
        ((n : Nat) => Lift.intro (motive (Int.neg_succ n)) (case_neg_succ n))
        target);

// x - y as an integer, by cases on whether y reaches x. This is the only
// interesting part of addition: it is what the mixed sign cases need.
Int.sub_nat_nat : Nat -> Nat -> Int :=
	(x : Nat) => (y : Nat) =>
	Nat.rec
		((t : Nat) => Int)
		(Int.nat (Nat.Sub x y))
		((k : Nat) => (ih : Int) => Int.neg_succ k)
		(Nat.Sub y x);

// Addition by cases on both arguments. The three equations below are now
// consequences of this rather than a partial specification of an opaque
// function - they hold by reflexivity in the checker, and stay assumptions only
// because the exporter cannot yet carry a proof that needs a recursor to fire.
Int.Add : Int -> Int -> Int :=
	(a : Int) => (b : Int) =>
	Int.rec
		((t : Int) => Int)
		((x : Nat) =>
			Int.rec
				((t : Int) => Int)
        // x + y
				((y : Nat) => Int.nat (Nat.Add x y))
        // x - s(y)
				((y : Nat) => Int.sub_nat_nat x (Nat.succ y))
				b)
		((x : Nat) =>
			Int.rec
				((t : Int) => Int)
        // -s(x) + y
				((y : Nat) => Int.sub_nat_nat y (Nat.succ x))
        // -s(x) - s(y) = -s(s(x + y))
				((y : Nat) => Int.neg_succ (Nat.succ (Nat.Add x y)))
				b)
		a;

Int.Add.add_nat_nat :
	(x : Nat) ->
	(y : Nat) ->
	Eq Int
		(Int.Add (Int.nat x) (Int.nat y))
		(Int.nat (Nat.Add x y)) :=
	(x : Nat) => (y : Nat) => Eq.refl Int (Int.nat (Nat.Add x y));
	
Int.Add.add_neg_succ_neg_succ :
	(x : Nat) ->
	(y : Nat) ->
	Eq Int
		(Int.Add (Int.neg_succ x) (Int.neg_succ y))
		(Int.neg_succ (Nat.succ (Nat.Add x y))) :=
	(x : Nat) => (y : Nat) =>
	Eq.refl Int (Int.neg_succ (Nat.succ (Nat.Add x y)));

Int.Add.add_zero_neg_succ:
	(x : Nat) ->
	Eq Int
		(Int.Add (Int.nat Nat.zero) (Int.neg_succ x))
		(Int.neg_succ x) :=
	(x : Nat) => Eq.refl Int (Int.neg_succ x);

// match x
// our motive is an int to a nat
// on case nat, it's just the enclosed value
// if it's neg_succ (-(x + 1)), it's x + 1

Int.abs_nat : Int -> Nat := 
(target : Int) =>
	Int.rec
		(Int => Nat)
		((x : Nat) => x)
		((x : Nat) => Nat.succ x)
	target;

abs_zero_eq_zero : Eq Nat (Int.abs_nat (Int.nat Nat.zero)) Nat.zero :=
	Eq.refl Nat Nat.zero;

abs_one_eq_one : Eq Nat (Int.abs_nat (Int.nat Nat.one)) Nat.one :=
	Eq.refl Nat Nat.one;
// -n, with -0 = 0 and -(succ k) = neg_succ k
Int.neg_nat : Nat -> Int :=
	(n : Nat) =>
	Nat.rec ((t : Nat) => Int)
		(Int.nat Nat.zero)
		((k : Nat) => (ih : Int) => Int.neg_succ k)
		n;

Int.abs_neg_nat : (n : Nat) -> Eq Nat (Int.abs_nat (Int.neg_nat n)) n :=
	Nat.rec_prop
		((n : Nat) => Eq Nat (Int.abs_nat (Int.neg_nat n)) n)
		(Eq.refl Nat Nat.zero)
		((k : Nat) => (ih : Eq Nat (Int.abs_nat (Int.neg_nat k)) k) =>
			Eq.refl Nat (Nat.succ k));

// x * d for a natural d: all the numerator of a sum of fractions needs
Int.mul_nat : Int -> Nat -> Int :=
	(x : Int) => (d : Nat) =>
	Int.rec ((t : Int) => Int)
		((m : Nat) => Int.nat (Nat.Mul m d))
		((m : Nat) => Int.neg_nat (Nat.Mul (Nat.succ m) d))
		x;
