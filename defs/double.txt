@require "arith.txt"

Nat.double : Nat -> Nat :=
	(n : Nat) =>
	Nat.rec
		((t : Nat) => Nat)
		Nat.zero
		((m : Nat) => (ih : Nat) => Nat.succ (Nat.succ ih))
		n;

// double is injective. Induction on the first argument with the second still
// quantified, then cases on the second.
Nat.double_inj.zero : (b : Nat) ->
	Eq Nat (Nat.double Nat.zero) (Nat.double b) -> Eq Nat Nat.zero b :=
	Nat.rec_prop
		((b : Nat) =>
			Eq Nat (Nat.double Nat.zero) (Nat.double b) -> Eq Nat Nat.zero b)
		((h : Eq Nat Nat.zero Nat.zero) => Eq.refl Nat Nat.zero)
		((b : Nat) =>
			(ih : Eq Nat (Nat.double Nat.zero) (Nat.double b) ->
				Eq Nat Nat.zero b) =>
			(h : Eq Nat Nat.zero (Nat.succ (Nat.succ (Nat.double b)))) =>
			False.elim (Eq Nat Nat.zero (Nat.succ b))
				(Nat.zero_ne_succ (Nat.succ (Nat.double b)) h));

Nat.double_inj.succ : (a : Nat) ->
	((b : Nat) -> Eq Nat (Nat.double a) (Nat.double b) -> Eq Nat a b) ->
	(b : Nat) ->
	Eq Nat (Nat.double (Nat.succ a)) (Nat.double b) -> Eq Nat (Nat.succ a) b :=
	(a : Nat) =>
	(ih : (b : Nat) -> Eq Nat (Nat.double a) (Nat.double b) -> Eq Nat a b) =>
	Nat.rec_prop
		((b : Nat) =>
			Eq Nat (Nat.double (Nat.succ a)) (Nat.double b) ->
			Eq Nat (Nat.succ a) b)
		((h : Eq Nat (Nat.succ (Nat.succ (Nat.double a))) Nat.zero) =>
			False.elim (Eq Nat (Nat.succ a) Nat.zero)
				(Nat.succ_ne_zero (Nat.succ (Nat.double a)) h))
		((b : Nat) =>
			(ihb : Eq Nat (Nat.double (Nat.succ a)) (Nat.double b) ->
				Eq Nat (Nat.succ a) b) =>
			(h : Eq Nat
				(Nat.succ (Nat.succ (Nat.double a)))
				(Nat.succ (Nat.succ (Nat.double b)))) =>
			Eq.congr Nat Nat Nat.succ a b
				(ih b
					(Nat.succ_inj (Nat.double a) (Nat.double b)
						(Nat.succ_inj
							(Nat.succ (Nat.double a))
							(Nat.succ (Nat.double b))
							h))));

Nat.double_inj : (a : Nat) -> (b : Nat) ->
	Eq Nat (Nat.double a) (Nat.double b) -> Eq Nat a b :=
	Nat.rec_prop
		((a : Nat) =>
			(b : Nat) -> Eq Nat (Nat.double a) (Nat.double b) -> Eq Nat a b)
		Nat.double_inj.zero
		Nat.double_inj.succ;

Nat.double_eq : (n : Nat) -> Eq Nat (Nat.double n) (Nat.Add n n) :=
	Nat.rec_prop
		((n : Nat) => Eq Nat (Nat.double n) (Nat.Add n n))
		(Eq.refl Nat Nat.zero)
		((n : Nat) => (ih : Eq Nat (Nat.double n) (Nat.Add n n)) =>
			Eq.trans Nat
				(Nat.succ (Nat.succ (Nat.double n)))
				(Nat.succ (Nat.succ (Nat.Add n n)))
				(Nat.Add (Nat.succ n) (Nat.succ n))
				(Eq.congr Nat Nat Nat.succ
					(Nat.succ (Nat.double n)) (Nat.succ (Nat.Add n n))
					(Eq.congr Nat Nat Nat.succ (Nat.double n) (Nat.Add n n) ih))
				(Eq.symm Nat
					(Nat.Add (Nat.succ n) (Nat.succ n))
					(Nat.succ (Nat.succ (Nat.Add n n)))
					(Eq.congr Nat Nat Nat.succ
						(Nat.Add (Nat.succ n) n) (Nat.succ (Nat.Add n n))
						(Nat.Add.succ_add n n))));

Nat.double_add : (p : Nat) -> (q : Nat) ->
	Eq Nat (Nat.double (Nat.Add p q)) (Nat.Add (Nat.double p) (Nat.double q)) :=
	(p : Nat) =>
	Nat.rec_prop
		((q : Nat) =>
			Eq Nat (Nat.double (Nat.Add p q)) (Nat.Add (Nat.double p) (Nat.double q)))
		(Eq.refl Nat (Nat.double p))
		((q : Nat) =>
			(ih : Eq Nat (Nat.double (Nat.Add p q))
				(Nat.Add (Nat.double p) (Nat.double q))) =>
			Eq.congr Nat Nat Nat.succ
				(Nat.succ (Nat.double (Nat.Add p q)))
				(Nat.succ (Nat.Add (Nat.double p) (Nat.double q)))
				(Eq.congr Nat Nat Nat.succ
					(Nat.double (Nat.Add p q))
					(Nat.Add (Nat.double p) (Nat.double q)) ih));

Nat.Mul.mul_double : (x : Nat) -> (y : Nat) ->
	Eq Nat (Nat.Mul x (Nat.double y)) (Nat.double (Nat.Mul x y)) :=
	(x : Nat) =>
	Nat.rec_prop
		((y : Nat) => Eq Nat (Nat.Mul x (Nat.double y)) (Nat.double (Nat.Mul x y)))
		(Eq.refl Nat Nat.zero)
		((y : Nat) =>
			(ih : Eq Nat (Nat.Mul x (Nat.double y)) (Nat.double (Nat.Mul x y))) =>
			Eq.trans Nat
				(Nat.Add x (Nat.Add x (Nat.Mul x (Nat.double y))))
				(Nat.Add x (Nat.Add x (Nat.double (Nat.Mul x y))))
				(Nat.double (Nat.Add x (Nat.Mul x y)))
				(Eq.congr Nat Nat (Nat.Add x)
					(Nat.Add x (Nat.Mul x (Nat.double y)))
					(Nat.Add x (Nat.double (Nat.Mul x y)))
					(Eq.congr Nat Nat (Nat.Add x)
						(Nat.Mul x (Nat.double y)) (Nat.double (Nat.Mul x y)) ih))
				(Eq.trans Nat
					(Nat.Add x (Nat.Add x (Nat.double (Nat.Mul x y))))
					(Nat.Add (Nat.Add x x) (Nat.double (Nat.Mul x y)))
					(Nat.double (Nat.Add x (Nat.Mul x y)))
					(Eq.symm Nat
						(Nat.Add (Nat.Add x x) (Nat.double (Nat.Mul x y)))
						(Nat.Add x (Nat.Add x (Nat.double (Nat.Mul x y))))
						(Nat.Add.add_assoc x x (Nat.double (Nat.Mul x y))))
					(Eq.trans Nat
						(Nat.Add (Nat.Add x x) (Nat.double (Nat.Mul x y)))
						(Nat.Add (Nat.double x) (Nat.double (Nat.Mul x y)))
						(Nat.double (Nat.Add x (Nat.Mul x y)))
						(Eq.congr Nat Nat
							((t : Nat) => Nat.Add t (Nat.double (Nat.Mul x y)))
							(Nat.Add x x) (Nat.double x)
							(Eq.symm Nat (Nat.double x) (Nat.Add x x) (Nat.double_eq x)))
						(Eq.symm Nat
							(Nat.double (Nat.Add x (Nat.Mul x y)))
							(Nat.Add (Nat.double x) (Nat.double (Nat.Mul x y)))
							(Nat.double_add x (Nat.Mul x y))))));

Nat.Mul.double_mul : (x : Nat) -> (y : Nat) ->
	Eq Nat (Nat.Mul (Nat.double x) y) (Nat.double (Nat.Mul x y)) :=
	(x : Nat) =>
	Nat.rec_prop
		((y : Nat) => Eq Nat (Nat.Mul (Nat.double x) y) (Nat.double (Nat.Mul x y)))
		(Eq.refl Nat Nat.zero)
		((y : Nat) =>
			(ih : Eq Nat (Nat.Mul (Nat.double x) y) (Nat.double (Nat.Mul x y))) =>
			Eq.trans Nat
				(Nat.Add (Nat.double x) (Nat.Mul (Nat.double x) y))
				(Nat.Add (Nat.double x) (Nat.double (Nat.Mul x y)))
				(Nat.double (Nat.Add x (Nat.Mul x y)))
				(Eq.congr Nat Nat (Nat.Add (Nat.double x))
					(Nat.Mul (Nat.double x) y) (Nat.double (Nat.Mul x y)) ih)
				(Eq.symm Nat
					(Nat.double (Nat.Add x (Nat.Mul x y)))
					(Nat.Add (Nat.double x) (Nat.double (Nat.Mul x y)))
					(Nat.double_add x (Nat.Mul x y))));

Nat.double_sq : (k : Nat) ->
	Eq Nat (Nat.Mul (Nat.double k) (Nat.double k))
		(Nat.double (Nat.double (Nat.Mul k k))) :=
	(k : Nat) =>
	Eq.trans Nat
		(Nat.Mul (Nat.double k) (Nat.double k))
		(Nat.double (Nat.Mul k (Nat.double k)))
		(Nat.double (Nat.double (Nat.Mul k k)))
		(Nat.Mul.double_mul k (Nat.double k))
		(Eq.congr Nat Nat Nat.double
			(Nat.Mul k (Nat.double k)) (Nat.double (Nat.Mul k k))
			(Nat.Mul.mul_double k k));

Nat.Add.two_add : (x : Nat) ->
	Eq Nat (Nat.Add Nat.two x) (Nat.succ (Nat.succ x)) :=
	(x : Nat) =>
	Eq.trans Nat
		(Nat.Add (Nat.succ (Nat.succ Nat.zero)) x)
		(Nat.succ (Nat.Add (Nat.succ Nat.zero) x))
		(Nat.succ (Nat.succ x))
		(Nat.Add.succ_add (Nat.succ Nat.zero) x)
		(Eq.congr Nat Nat Nat.succ
			(Nat.Add (Nat.succ Nat.zero) x) (Nat.succ x)
			(Eq.trans Nat
				(Nat.Add (Nat.succ Nat.zero) x)
				(Nat.succ (Nat.Add Nat.zero x))
				(Nat.succ x)
				(Nat.Add.succ_add Nat.zero x)
				(Eq.congr Nat Nat Nat.succ (Nat.Add Nat.zero x) x
					(Nat.Add.zero_add x))));

Nat.Mul.two_mul : (n : Nat) -> Eq Nat (Nat.Mul Nat.two n) (Nat.double n) :=
	Nat.rec_prop
		((n : Nat) => Eq Nat (Nat.Mul Nat.two n) (Nat.double n))
		(Eq.refl Nat Nat.zero)
		((n : Nat) => (ih : Eq Nat (Nat.Mul Nat.two n) (Nat.double n)) =>
			Eq.trans Nat
				(Nat.Add Nat.two (Nat.Mul Nat.two n))
				(Nat.Add Nat.two (Nat.double n))
				(Nat.succ (Nat.succ (Nat.double n)))
				(Eq.congr Nat Nat (Nat.Add Nat.two)
					(Nat.Mul Nat.two n) (Nat.double n) ih)
				(Nat.Add.two_add (Nat.double n)));
