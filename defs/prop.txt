@inductive
True : Prop;
@intro(True)
True.triv : True;
@inductive
False : Prop;
// no constructors, so no cases: from a False, anything. Eliminating a Prop
// into Type is allowed here because False has no constructor that could carry
// data to reveal.
@rec(False)
False.rec :
	(motive : False -> Type) ->
	(target : False) ->
	motive target;

@inductive
Lift : Prop -> Type;
@intro(Lift)
Lift.intro : (p : Prop) -> p -> Lift p;
@axiom
Lift.elim : (p : Prop) -> Lift p -> p;

False.rec_prop :
	(motive : False -> Prop) ->
	(target : False) ->
	motive target :=
	(motive : False -> Prop) =>
	(target : False) =>
	Lift.elim
		(motive target)
		(False.rec ((x : False) => Lift (motive x)) target);

False.elim : (T : Prop) -> False -> T :=
	(T : Prop) => (f : False) => False.rec_prop ((x : False) => T) f;

@inductive
Prop.Iff : (a : Prop) -> (b : Prop) -> Prop;
@intro(Prop.Iff)
Prop.Iff.intro : (a : Prop) -> (b : Prop) -> (a -> b) -> (b -> a) -> Prop.Iff a b;

@inductive
Not : Prop -> Prop;
@intro(Not)
Not.intro : (a : Prop) -> (a -> False) -> Not a;
@rec(Not)
Not.rec :
	(a : Prop) ->
	(motive : Not a -> Type) ->
	(case_intro : (proof : a -> False) -> motive (Not.intro a proof)) ->
	(evidence : Not a) ->
	motive evidence;

// Not.rec eliminates into Type; Lift it to eliminate into Prop, the same way
// Nat.rec_prop and Eq.rec_prop do.
Not.rec_prop :
	(a : Prop) ->
	(motive : Not a -> Prop) ->
	(case_intro : (proof : a -> False) -> motive (Not.intro a proof)) ->
	(evidence : Not a) ->
	motive evidence :=
	(a : Prop) =>
	(motive : Not a -> Prop) =>
	(case_intro : (proof : a -> False) -> motive (Not.intro a proof)) =>
	(evidence : Not a) =>
	Lift.elim
		(motive evidence)
		(Not.rec
			a
			((e : Not a) => Lift (motive e))
			((proof : a -> False) =>
				Lift.intro
					(motive (Not.intro a proof))
					(case_intro proof))
			evidence);

Not.elim : (a : Prop) -> Not a -> a -> False :=
	(a : Prop) => (p : Not a) =>
	Not.rec_prop
		a
		(Not a => (a -> False))
		((proof : a -> False) => proof)
		p;

Prop.Or.elim : (a : Prop) -> (b : Prop) -> (c : Prop) -> Prop.Or a b -> (a -> c) -> (b -> c) -> c;
