@axiom
Not : Prop -> Prop;
@axiom
Not.intro : (a : Prop) -> (a -> False) -> Not a;
@axiom
Not.rec :
	(a : Prop) ->
	(motive : Not a -> Type) ->
	(case_intro : (proof : a -> False) -> motive (Not.intro a proof)) ->
	(evidence : Not a) ->
	motive evidence;

// Not.rec eliminates into Type; Lift it to eliminate into Prop, the same way
// Nat.rec_prop and Eq.rec_prop do.
Not.rec_prop :
	(a : Prop) ->
	(motive : Not a -> Prop) ->
	(case_intro : (proof : a -> False) -> motive (Not.intro a proof)) ->
	(evidence : Not a) ->
	motive evidence :=
	(a : Prop) =>
	(motive : Not a -> Prop) =>
	(case_intro : (proof : a -> False) -> motive (Not.intro a proof)) =>
	(evidence : Not a) =>
	Lift.elim
		(motive evidence)
		(Not.rec
			a
			((e : Not a) => Lift (motive e))
			((proof : a -> False) =>
				Lift.intro
					(motive (Not.intro a proof))
					(case_intro proof))
			evidence);

Not.elim : (a : Prop) -> Not a -> a -> False :=
	(a : Prop) => (p : Not a) =>
	Not.rec_prop
		a
		(Not a => (a -> False))
		((proof : a -> False) => proof)
		p;
