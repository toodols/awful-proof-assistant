@inductive
False : Prop;

// into Type is allowed here because False has no constructor that could carry
// data to reveal.
@rec(False)
False.rec :
	(motive : False -> Type) ->
	(target : False) ->

@inductive
Lift : Prop -> Type;
@intro(Lift)
Lift.intro : (p : Prop) -> p -> Lift p;
@rec(Lift)
Lift.rec :
	(p : Prop) ->
	(motive : Lift p -> Type) ->
	(case_intro : (x : p) -> motive (Lift.intro p x)) ->
	(target : Lift p) ->
	motive target;

@iota(Lift)
Lift.rec p motive case_intro (Lift.intro p x) :
	motive (Lift.intro p x) := case_intro x;

// Lift p is a Type, so eliminating it into Prop is small elimination and needs
// no permission. Unlike every other rec_prop here, this one is stated rather
// than derived: deriving one goes through Lift, which cannot work for Lift.
@rec(Lift)
Lift.rec_prop :
	(p : Prop) ->
	(motive : Lift p -> Prop) ->
	(case_intro : (x : p) -> motive (Lift.intro p x)) ->
	(target : Lift p) ->
	motive target;

@iota(Lift)
Lift.rec_prop p motive case_intro (Lift.intro p x) :
	motive (Lift.intro p x) := case_intro x;

Lift.elim : (p : Prop) -> Lift p -> p :=
	(p : Prop) => (target : Lift p) =>
	Lift.rec_prop p ((x : Lift p) => p) ((x : p) => x) target;

False.rec_prop :
	(motive : False -> Prop) ->
	(target : False) ->
	motive target :=
	(motive : False -> Prop) =>
	(target : False) =>
	Lift.elim
		(motive target)
		(False.rec ((x : False) => Lift (motive x)) target);

False.elim : (T : Prop) -> False -> T :=
	(T : Prop) => (f : False) => False.rec_prop ((x : False) => T) f;

@inductive
Prop.Iff : (a : Prop) -> (b : Prop) -> Prop;
@intro(Prop.Iff)
Prop.Iff.intro : (a : Prop) -> (b : Prop) -> (a -> b) -> (b -> a) -> Prop.Iff a b;
@rec(Prop.Iff)
Prop.Iff.rec :
	(a : Prop) -> (b : Prop) ->
	(motive : Prop.Iff a b -> Type) ->
	(case_intro : (mp : a -> b) -> (mpr : b -> a) ->
		motive (Prop.Iff.intro a b mp mpr)) ->
	(target : Prop.Iff a b) ->
	motive target;

@iota(Prop.Iff)
Prop.Iff.rec a b motive case_intro (Prop.Iff.intro a b mp mpr) :
	motive (Prop.Iff.intro a b mp mpr) := case_intro mp mpr;

Prop.Iff.rec_prop :
	(a : Prop) -> (b : Prop) ->
	(motive : Prop.Iff a b -> Prop) ->
	(case_intro : (mp : a -> b) -> (mpr : b -> a) ->
		motive (Prop.Iff.intro a b mp mpr)) ->
	(target : Prop.Iff a b) ->
	motive target :=
	(a : Prop) => (b : Prop) =>
	(motive : Prop.Iff a b -> Prop) =>
	(case_intro : (mp : a -> b) -> (mpr : b -> a) ->
		motive (Prop.Iff.intro a b mp mpr)) =>
	(target : Prop.Iff a b) =>
	Lift.elim
		(motive target)
		(Prop.Iff.rec a b
			((t : Prop.Iff a b) => Lift (motive t))
			((mp : a -> b) => (mpr : b -> a) =>
				Lift.intro
					(motive (Prop.Iff.intro a b mp mpr))
					(case_intro mp mpr))
			target);

Prop.Iff.mp : (a : Prop) -> (b : Prop) -> Prop.Iff a b -> a -> b :=
	(a : Prop) => (b : Prop) => (h : Prop.Iff a b) =>
	Prop.Iff.rec_prop a b
		((t : Prop.Iff a b) => a -> b)
		((mp : a -> b) => (mpr : b -> a) => mp)
		h;

Prop.Iff.mpr : (a : Prop) -> (b : Prop) -> Prop.Iff a b -> b -> a :=
	(a : Prop) => (b : Prop) => (h : Prop.Iff a b) =>
	Prop.Iff.rec_prop a b
		((t : Prop.Iff a b) => b -> a)
		((mp : a -> b) => (mpr : b -> a) => mpr)
		h;

Prop.Not : Prop -> Prop := (a : Prop) => a -> False;

@inductive
Decidable : (a : Prop) -> Type;

@intro(Decidable)
Decidable.yes : (a : Prop) -> (h : a) -> Decidable a;

@intro(Decidable)
Decidable.no : (a : Prop) -> (h : Prop.Not a) -> Decidable a;

// `a` is a parameter, not an index: both constructors take it first and hand it
// straight back, so it comes before the motive and the motive does not mention
// it.
@rec(Decidable)
Decidable.rec :
	(a : Prop) ->
	(motive : Decidable a -> Type) ->
	(case_yes : (h : a) -> motive (Decidable.yes a h)) ->
	(case_no : (h : Prop.Not a) -> motive (Decidable.no a h)) ->
	(target : Decidable a) ->
	motive target;

@iota(Decidable)
Decidable.rec a motive case_yes case_no (Decidable.yes a h) :
	motive (Decidable.yes a h) := case_yes h;
@iota(Decidable)
Decidable.rec a motive case_yes case_no (Decidable.no a h) :
	motive (Decidable.no a h) := case_no h;

// Reducing a fraction needs a greatest common divisor and exact division.
// There is no choice principle here, but `classical` decides every proposition,
// and a decision can be eliminated into Nat. So both are found by a bounded
// search that asks `classical` at each step, and are specified by the lemmas
// below rather than by computation.

Decidable.rec_prop :
	(a : Prop) ->
	(motive : Decidable a -> Prop) ->
	(case_yes : (h : a) -> motive (Decidable.yes a h)) ->
	(case_no : (h : Prop.Not a) -> motive (Decidable.no a h)) ->
	(target : Decidable a) ->
	motive target :=
	(a : Prop) =>
	(motive : Decidable a -> Prop) =>
	(case_yes : (h : a) -> motive (Decidable.yes a h)) =>
	(case_no : (h : Prop.Not a) -> motive (Decidable.no a h)) =>
	(target : Decidable a) =>
	Lift.elim (motive target)
		(Decidable.rec a
			((t : Decidable a) => Lift (motive t))
			((h : a) => Lift.intro (motive (Decidable.yes a h)) (case_yes h))
			((h : Prop.Not a) => Lift.intro (motive (Decidable.no a h)) (case_no h))
			target);

@axiom
classical : (a : Prop) -> Decidable a;

@inductive
Prop.And : Prop -> Prop -> Prop;
@intro(Prop.And)
Prop.And.intro : (a : Prop) -> (b : Prop) -> a -> b -> Prop.And a b;
@rec(Prop.And)
Prop.And.rec :
	(a : Prop) -> (b : Prop) ->
	(motive : Prop.And a b -> Type) ->
	(case_intro : (x : a) -> (y : b) -> motive (Prop.And.intro a b x y)) ->
	(target : Prop.And a b) ->
	motive target;

@iota(Prop.And)
Prop.And.rec a b motive case_intro (Prop.And.intro a b x y) :
	motive (Prop.And.intro a b x y) := case_intro x y;

Prop.And.rec_prop :
	(a : Prop) -> (b : Prop) ->
	(motive : Prop.And a b -> Prop) ->
	(case_intro : (x : a) -> (y : b) -> motive (Prop.And.intro a b x y)) ->
	(target : Prop.And a b) ->
	motive target :=
	(a : Prop) => (b : Prop) =>
	(motive : Prop.And a b -> Prop) =>
	(case_intro : (x : a) -> (y : b) -> motive (Prop.And.intro a b x y)) =>
	(target : Prop.And a b) =>
	Lift.elim
		(motive target)
		(Prop.And.rec a b
			((t : Prop.And a b) => Lift (motive t))
			((x : a) => (y : b) =>
				Lift.intro
					(motive (Prop.And.intro a b x y))
					(case_intro x y))
			target);

Prop.And.left : (a : Prop) -> (b : Prop) -> Prop.And a b -> a :=
	(a : Prop) => (b : Prop) => (h : Prop.And a b) =>
	Prop.And.rec_prop a b
		((t : Prop.And a b) => a)
		((x : a) => (y : b) => x)
		h;

Prop.And.right : (a : Prop) -> (b : Prop) -> Prop.And a b -> b :=
	(a : Prop) => (b : Prop) => (h : Prop.And a b) =>
	Prop.And.rec_prop a b
		((t : Prop.And a b) => b)
		((x : a) => (y : b) => y)
		h;

@inductive
Prop.Or : Prop -> Prop -> Prop;
@intro(Prop.Or)
Prop.Or.inl : (a : Prop) -> (b : Prop) -> a -> Prop.Or a b;
@intro(Prop.Or)
Prop.Or.inr : (a : Prop) -> (b : Prop) -> b -> Prop.Or a b;
// Or has two constructors, so it may only eliminate into Prop - which is all
// elim ever wanted.
@rec(Prop.Or)
Prop.Or.rec_prop :
	(a : Prop) -> (b : Prop) ->
	(motive : Prop.Or a b -> Prop) ->
	(case_inl : (x : a) -> motive (Prop.Or.inl a b x)) ->
	(case_inr : (y : b) -> motive (Prop.Or.inr a b y)) ->
	(target : Prop.Or a b) ->
	motive target;

@iota(Prop.Or)
Prop.Or.rec_prop a b motive case_inl case_inr (Prop.Or.inl a b x) :
	motive (Prop.Or.inl a b x) := case_inl x;
@iota(Prop.Or)
Prop.Or.rec_prop a b motive case_inl case_inr (Prop.Or.inr a b y) :
	motive (Prop.Or.inr a b y) := case_inr y;

Prop.Or.elim : (a : Prop) -> (b : Prop) -> (c : Prop) -> Prop.Or a b -> (a -> c) -> (b -> c) -> c :=
	(a : Prop) => (b : Prop) => (c : Prop) =>
	(h : Prop.Or a b) => (l : a -> c) => (r : b -> c) =>
	Prop.Or.rec_prop a b
		((t : Prop.Or a b) => c)
		((x : a) => l x)
		((y : b) => r y)
		h;

@inductive
Exists : (A : Type) -> (P : A -> Prop) -> Prop;
@intro(Exists)
Exists.intro : (A : Type) -> (P : A -> Prop) -> (a : A) -> P a -> Exists A P;
@rec(Exists)
Exists.rec_prop :
	(A : Type) -> (P : A -> Prop) ->
	(motive : Exists A P -> Prop) ->
	(case_intro : (a : A) -> (h : P a) -> motive (Exists.intro A P a h)) ->
	(target : Exists A P) ->
	motive target;

@iota(Exists)
Exists.rec_prop A P motive case_intro (Exists.intro A P a h) :
	motive (Exists.intro A P a h) := case_intro a h;

Exists.elim : (A : Type) -> (P : A -> Prop) -> (C : Prop) ->
	Exists A P -> ((a : A) -> P a -> C) -> C :=
	(A : Type) => (P : A -> Prop) => (C : Prop) =>
	(h : Exists A P) => (k : (a : A) -> P a -> C) =>
	Exists.rec_prop A P
		((t : Exists A P) => C)
		((a : A) => (x : P a) => k a x)
		h;

@inductive
ForAll : (A : Type) -> (P : A -> Prop) -> Prop;
@intro(ForAll)
ForAll.intro : (A : Type) -> (P : A -> Prop) -> ((a : A) -> P a) -> ForAll A P;
@rec(ForAll)
ForAll.rec :
	(A : Type) -> (P : A -> Prop) ->
	(motive : ForAll A P -> Type) ->
	(case_intro : (f : (a : A) -> P a) -> motive (ForAll.intro A P f)) ->
	(target : ForAll A P) ->
	motive target;

@iota(ForAll)
ForAll.rec A P motive case_intro (ForAll.intro A P f) :
	motive (ForAll.intro A P f) := case_intro f;

ForAll.rec_prop :
	(A : Type) -> (P : A -> Prop) ->
	(motive : ForAll A P -> Prop) ->
	(case_intro : (f : (a : A) -> P a) -> motive (ForAll.intro A P f)) ->
	(target : ForAll A P) ->
	motive target :=
	(A : Type) => (P : A -> Prop) =>
	(motive : ForAll A P -> Prop) =>
	(case_intro : (f : (a : A) -> P a) -> motive (ForAll.intro A P f)) =>
	(target : ForAll A P) =>
	Lift.elim
		(motive target)
		(ForAll.rec A P
			((t : ForAll A P) => Lift (motive t))
			((f : (a : A) -> P a) =>
				Lift.intro (motive (ForAll.intro A P f)) (case_intro f))
			target);

ForAll.elim : (A : Type) -> (P : A -> Prop) -> ForAll A P -> (a : A) -> P a :=
	(A : Type) => (P : A -> Prop) => (h : ForAll A P) => (a : A) =>
	ForAll.rec_prop A P
		((t : ForAll A P) => P a)
		((f : (x : A) -> P x) => f a)
		h;
