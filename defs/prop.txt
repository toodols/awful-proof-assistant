@inductive
True : Prop;
@intro(True)
True.triv : True;
@inductive
False : Prop;
// no constructors, so no cases: from a False, anything. Eliminating a Prop
// into Type is allowed here because False has no constructor that could carry
// data to reveal.
@rec(False)
False.rec :
	(motive : False -> Type) ->
	(target : False) ->
	motive target;

@inductive
Lift : Prop -> Type;
@intro(Lift)
Lift.intro : (p : Prop) -> p -> Lift p;
// Lift p is a Type, so eliminating it into Prop is small elimination and needs
// no permission. The Prop motive is stated directly rather than derived from a
// Type one: the usual detour through Lift cannot be used for Lift itself.
@rec(Lift)
Lift.rec_prop :
	(p : Prop) ->
	(motive : Lift p -> Prop) ->
	(case_intro : (x : p) -> motive (Lift.intro p x)) ->
	(target : Lift p) ->
	motive target;

Lift.elim : (p : Prop) -> Lift p -> p :=
	(p : Prop) => (target : Lift p) =>
	Lift.rec_prop p ((x : Lift p) => p) ((x : p) => x) target;

False.rec_prop :
	(motive : False -> Prop) ->
	(target : False) ->
	motive target :=
	(motive : False -> Prop) =>
	(target : False) =>
	Lift.elim
		(motive target)
		(False.rec ((x : False) => Lift (motive x)) target);

False.elim : (T : Prop) -> False -> T :=
	(T : Prop) => (f : False) => False.rec_prop ((x : False) => T) f;

@inductive
Prop.Iff : (a : Prop) -> (b : Prop) -> Prop;
@intro(Prop.Iff)
Prop.Iff.intro : (a : Prop) -> (b : Prop) -> (a -> b) -> (b -> a) -> Prop.Iff a b;

@inductive
Not : Prop -> Prop;
@intro(Not)
Not.intro : (a : Prop) -> (a -> False) -> Not a;
@rec(Not)
Not.rec :
	(a : Prop) ->
	(motive : Not a -> Type) ->
	(case_intro : (proof : a -> False) -> motive (Not.intro a proof)) ->
	(evidence : Not a) ->
	motive evidence;

// Not.rec eliminates into Type; Lift it to eliminate into Prop, the same way
// Nat.rec_prop and Eq.rec_prop do.
Not.rec_prop :
	(a : Prop) ->
	(motive : Not a -> Prop) ->
	(case_intro : (proof : a -> False) -> motive (Not.intro a proof)) ->
	(evidence : Not a) ->
	motive evidence :=
	(a : Prop) =>
	(motive : Not a -> Prop) =>
	(case_intro : (proof : a -> False) -> motive (Not.intro a proof)) =>
	(evidence : Not a) =>
	Lift.elim
		(motive evidence)
		(Not.rec
			a
			((e : Not a) => Lift (motive e))
			((proof : a -> False) =>
				Lift.intro
					(motive (Not.intro a proof))
					(case_intro proof))
			evidence);

Not.elim : (a : Prop) -> Not a -> a -> False :=
	(a : Prop) => (p : Not a) =>
	Not.rec_prop
		a
		(Not a => (a -> False))
		((proof : a -> False) => proof)
		p;

@inductive
Prop.And : Prop -> Prop -> Prop;
@intro(Prop.And)
Prop.And.intro : (a : Prop) -> (b : Prop) -> a -> b -> Prop.And a b;
@rec(Prop.And)
Prop.And.rec_prop :
	(a : Prop) -> (b : Prop) ->
	(motive : Prop.And a b -> Prop) ->
	(case_intro : (x : a) -> (y : b) -> motive (Prop.And.intro a b x y)) ->
	(target : Prop.And a b) ->
	motive target;

Prop.And.left : (a : Prop) -> (b : Prop) -> Prop.And a b -> a :=
	(a : Prop) => (b : Prop) => (h : Prop.And a b) =>
	Prop.And.rec_prop a b
		((t : Prop.And a b) => a)
		((x : a) => (y : b) => x)
		h;

Prop.And.right : (a : Prop) -> (b : Prop) -> Prop.And a b -> b :=
	(a : Prop) => (b : Prop) => (h : Prop.And a b) =>
	Prop.And.rec_prop a b
		((t : Prop.And a b) => b)
		((x : a) => (y : b) => y)
		h;

@inductive
Prop.Or : Prop -> Prop -> Prop;
@intro(Prop.Or)
Prop.Or.inl : (a : Prop) -> (b : Prop) -> a -> Prop.Or a b;
@intro(Prop.Or)
Prop.Or.inr : (a : Prop) -> (b : Prop) -> b -> Prop.Or a b;
// Or has two constructors, so it may only eliminate into Prop - which is all
// elim ever wanted.
@rec(Prop.Or)
Prop.Or.rec_prop :
	(a : Prop) -> (b : Prop) ->
	(motive : Prop.Or a b -> Prop) ->
	(case_inl : (x : a) -> motive (Prop.Or.inl a b x)) ->
	(case_inr : (y : b) -> motive (Prop.Or.inr a b y)) ->
	(target : Prop.Or a b) ->
	motive target;

Prop.Or.elim : (a : Prop) -> (b : Prop) -> (c : Prop) -> Prop.Or a b -> (a -> c) -> (b -> c) -> c :=
	(a : Prop) => (b : Prop) => (c : Prop) =>
	(h : Prop.Or a b) => (l : a -> c) => (r : b -> c) =>
	Prop.Or.rec_prop a b
		((t : Prop.Or a b) => c)
		((x : a) => l x)
		((y : b) => r y)
		h;
